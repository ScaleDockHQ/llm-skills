import pytest
from deepeval import assert_test
from deepeval.dataset import EvaluationDataset, Golden
from deepeval.metrics import ExactMatchMetric, GEval
from deepeval.test_case import LLMTestCase, SingleTurnParams

dataset = EvaluationDataset(goldens=[
    Golden(input="What is 17 + 25? Reply with the number only.", expected_output="42"),
    Golden(input="What is the capital of France? Answer with one word.", expected_output="Paris"),
])

correctness = GEval(
    name="Correctness",
    criteria="Is the actual output correct according to the expected output?",
    evaluation_params=[SingleTurnParams.ACTUAL_OUTPUT, SingleTurnParams.EXPECTED_OUTPUT],
    threshold=0.5,
)


def my_app(question: str) -> str:
    """Replace with a call to the model or service under test."""
    return "42" if "17" in question else "Paris"


@pytest.mark.parametrize("golden", dataset.goldens)
def test_app(golden: Golden):
    case = LLMTestCase(
        input=golden.input,
        actual_output=my_app(golden.input),
        expected_output=golden.expected_output,
    )
    # Deterministic check first, judged check second.
    assert_test(test_case=case, metrics=[ExactMatchMetric(), correctness])
